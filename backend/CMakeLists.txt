cmake_minimum_required(VERSION 3.16)
project(campus_runner_backend LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(cr_sqlite vendor/sqlite3.c)
target_include_directories(cr_sqlite PUBLIC vendor)
target_compile_definitions(cr_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_FOREIGN_KEYS=1 SQLITE_OMIT_LOAD_EXTENSION)
add_library(cr_json vendor/cJSON.c)
target_include_directories(cr_json PUBLIC vendor)
target_compile_definitions(cr_json PRIVATE CJSON_NESTING_LIMIT=32)
add_library(cr_http vendor/civetweb/src/civetweb.c)
target_include_directories(cr_http PUBLIC vendor/civetweb/include)
target_compile_definitions(cr_http PRIVATE NO_SSL NO_CGI NO_FILES NO_CACHING)
target_link_libraries(cr_http PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(campus_runner_core src/campus_runner.c src/storage.c src/file_io.c)
target_include_directories(campus_runner_core PUBLIC include)
if(MSVC)
  target_compile_options(campus_runner_core PRIVATE /W4 /utf-8)
  target_compile_definitions(campus_runner_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(campus_runner_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(campus_runner_core PUBLIC m)
endif()
add_executable(campus_runner src/main.c)
target_link_libraries(campus_runner PRIVATE campus_runner_core)
add_executable(campus_server src/server_main.c src/api.c src/auth.c src/auth_api.c src/database.c src/runs_api.c src/catalog.c src/verification.c)
target_link_libraries(campus_server PRIVATE campus_runner_core cr_sqlite cr_json cr_http)
if(WIN32)
  target_link_libraries(campus_server PRIVATE bcrypt ws2_32)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(campus_server PRIVATE OpenSSL::Crypto Threads::Threads m)
  target_compile_definitions(campus_server PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(MSVC)
  target_compile_options(campus_runner PRIVATE /W4 /utf-8)
  target_compile_definitions(campus_runner PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(campus_server PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(campus_server PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(cr_http PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(campus_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_backend tests/test_backend.c)
  target_link_libraries(test_backend PRIVATE campus_runner_core)
  if(MSVC)
    target_compile_options(test_backend PRIVATE /W4 /utf-8 /UNDEBUG)
    target_compile_definitions(test_backend PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(test_backend PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  endif()
  add_test(NAME backend COMMAND test_backend)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:campus_runner>)
    add_test(NAME server COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server.py $<TARGET_FILE:campus_server>)
  endif()
  find_program(CR_NODE_EXECUTABLE NAMES node nodejs)
  if(CR_NODE_EXECUTABLE)
    add_test(NAME verification_admin COMMAND ${CR_NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_verification_admin.cjs)
  endif()
endif()
